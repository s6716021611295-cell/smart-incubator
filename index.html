Protocol: JSON HTTP POST / WebSocket

// ============================================================
// SMART INCUBATOR ESP32 / Arduino WiFi Firmware
// Hardware: Arduino UNO / ESP32, DHT22, Ultrasonic Fogger,
// Relay 220V (Heater), Relay 12V (Fan), Motor Turner, Buzzer
// ============================================================

#include <WiFi.h>
#include <HTTPClient.h>
#include "DHT.h"

#define DHTPIN 4
#define DHTTYPE DHT22

#define RELAY_HEATER 16   // Relay 1: Heater (220V)
#define RELAY_FOGGER 17   // Relay 2: Ultrasonic Maker
#define RELAY_FAN    18   // Relay 3: 12V Air Fan
#define MOTOR_PIN    19   // Motor Sprocket Egg Turner
#define BUZZER_PIN   23   // Alarm Buzzer

const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";
const char* serverUrl = "https://your-incubator-api.com/api/telemetry";

DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(115200);
  pinMode(RELAY_HEATER, OUTPUT);
  pinMode(RELAY_FOGGER, OUTPUT);
  pinMode(RELAY_FAN, OUTPUT);
  pinMode(MOTOR_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);

  dht.begin();
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi Connected!");
}

void loop() {
  float temp = dht.readTemperature();
  float hum = dht.readHumidity();

  if (isnan(temp) || isnan(hum)) {
    Serial.println("Failed to read from DHT sensor!");
    return;
  }

  // Automatic Incubator Control Logic
  if (temp < 37.3) {
    digitalWrite(RELAY_HEATER, HIGH); // Turn ON Heater
  } else if (temp > 37.7) {
    digitalWrite(RELAY_HEATER, LOW);  // Turn OFF Heater
  }

  if (hum < 55.0) {
    digitalWrite(RELAY_FOGGER, HIGH); // Turn ON Fogger
  } else if (hum > 65.0) {
    digitalWrite(RELAY_FOGGER, LOW);  // Turn OFF Fogger
  }

  // Send Telemetry to Web Dashboard
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(serverUrl);
    http.addHeader("Content-Type", "application/json");

    String json = "{\"temperature\":" + String(temp) + 
                  ",\"humidity\":" + String(hum) + "}";
    int httpResponseCode = http.POST(json);
    http.end();
  }

  delay(2000);
}
